cmake_minimum_required(VERSION 3.20)
project(encore_sendspin_lab LANGUAGES C CXX)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SENDSPIN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_OPUS OFF CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_ARTWORK OFF CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_VISUALIZER OFF CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_COLOR OFF CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_SOURCE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SENDSPIN_SOURCE_DIR}" sendspin)
add_executable(encore-sendspin main.cpp "${SENDSPIN_SOURCE_DIR}/examples/common/file_persistence_provider.cpp")
target_include_directories(encore-sendspin PRIVATE "${SENDSPIN_SOURCE_DIR}/examples/common")
target_link_libraries(encore-sendspin PRIVATE sendspin)
target_compile_features(encore-sendspin PRIVATE cxx_std_20)
