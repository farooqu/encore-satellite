cmake_minimum_required(VERSION 3.20)
project(encore_sendspin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(role METADATA COLOR ARTWORK VISUALIZER SOURCE OPUS)
  set(SENDSPIN_ENABLE_${role} OFF CACHE BOOL "" FORCE)
endforeach()
set(SENDSPIN_ENABLE_PLAYER ON CACHE BOOL "" FORCE)
set(SENDSPIN_ENABLE_CONTROLLER ON CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SENDSPIN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT SDK_SOURCE)
  message(FATAL_ERROR "Use packaging/sendspin.sh: SDK_SOURCE must be the locked source")
endif()
add_subdirectory(${SDK_SOURCE} sdk)
add_executable(encore-sendspin main.cpp)
target_link_libraries(encore-sendspin PRIVATE sendspin)
target_compile_options(encore-sendspin PRIVATE -Wall -Wextra -Werror)
if(ENCORE_STATIC)
  target_link_options(encore-sendspin PRIVATE -static)
endif()
